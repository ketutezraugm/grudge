    <script>
      window.__timelines = window.__timelines || {};

      // Exact identity eases (IDENTITY.md). No springs, no overshoot, no back.out.
      gsap.registerPlugin(CustomEase);
      CustomEase.create("enter", "0.2,0,0,1");
      CustomEase.create("exit", "0.4,0,1,1");
      CustomEase.create("standard", "0.2,0,0.2,1");

      // Motion tokens, in seconds
      var INSTANT = 0.1;
      var QUICK = 0.2;
      var DELIBERATE = 0.4;

      // Frame number at 30 fps -> seconds
      function F(n) {
        return n / 30;
      }

      var tl = gsap.timeline({ paused: true });

      // === CONCEPT A: PLOTTED ===
      // Match cut "empty-mark": frame 0 is the hook's last frame with the footage cut away.
      // The empty wrong-call mark sits where Polymarket's newest record mark was
      // (880.8, 708.8 under the hook's 1.6x push-in), at record-row size.

      // f0-12: the empty mark scales to mark size and travels to centre-left (deliberate, enter)
      tl.fromTo(
        "#mark",
        { x: 188, y: 124.8, scale: 0.1667, transformOrigin: "140.8px 144px" },
        { x: 0, y: 0, scale: 1, duration: DELIBERATE, ease: "enter" },
        F(0),
      );

      // f12-30: cover, by eighths. Six snaps, one eighth per 3 frames, stopping at three-quarters.
      // Sound: one dry pen tick per snap.
      tl.set("#m-w1", { opacity: 1 }, F(12));
      tl.set("#m-w2", { opacity: 1 }, F(15));
      tl.set("#m-w3", { opacity: 1 }, F(18));
      tl.set("#m-w4", { opacity: 1 }, F(21));
      tl.set("#m-w5", { opacity: 1 }, F(24));
      tl.set("#m-wedge", { opacity: 1 }, F(27));
      tl.set(".okta", { opacity: 0 }, F(27)); // the mark's own wedge takes over from the steps

      // f30-42: wind direction. The staff extends out of the circle along its angle (stroke offset, quick, enter)
      tl.to("#m-staff", { strokeDashoffset: 18.01, duration: QUICK, ease: "enter" }, F(30));

      // f42-54: speed. Barb, barb, half barb, 4 frames apart, each one instant snap.
      // Sound: a slightly heavier tick per barb.
      tl.to("#m-staff", { strokeDashoffset: 0, duration: INSTANT, ease: "enter" }, F(42));
      tl.to("#m-barb2", { strokeDashoffset: 0, duration: INSTANT, ease: "enter" }, F(46));
      tl.to("#m-barb3", { strokeDashoffset: 0, duration: INSTANT, ease: "enter" }, F(50));

      // f54-72: the wordmark arrives by axis: wdth 75 to 112.5 and wght 200 to 800 together (deliberate, enter).
      // Sound: silence.
      tl.fromTo(
        "#word",
        { "--wdth": 75, "--wght": 200, opacity: 0 },
        { "--wdth": 112.5, "--wght": 800, opacity: 1, duration: DELIBERATE, ease: "enter" },
        F(54),
      );

      // f72-90: the motto sets under it (opacity only, quick). Music enters on the hold.
      tl.fromTo("#motto", { opacity: 0 }, { opacity: 1, duration: QUICK, ease: "enter" }, F(72));

      // Hold to 4 s (frame 120)
      tl.set({}, {}, 4);

      window.__timelines["main"] = tl;
    </script>
